7. Map (Dictionary)

7.1 Apa Itu Map?

Map adalah struktur data yang menyimpan pasangan key-value (kunci-nilai).

Karakteristik Map:

  • Setiap key harus unik
  • Key dan value bisa berbagai tipe data
  • Tidak berurutan (unordered)
  • Dinamis (bisa bertambah/berkurang)

7.2 Deklarasi Map

package main import "fmt" func main() { // Cara 1: Dengan make mahasiswa := make(map[string]int) mahasiswa["Budi"] = 85 mahasiswa["Ani"] = 90 mahasiswa["Citra"] = 88 fmt.Println(mahasiswa) // Cara 2: Dengan map literal buah := map[string]int{ "Apel": 5000, "Mangga": 8000, "Jeruk": 3000, } fmt.Println(buah) }

7.3 Mengakses dan Mengubah Map

package main import "fmt" func main() { nilai := map[string]int{ "Matematika": 85, "Fisika": 90, "Kimia": 88, } // Akses nilai fmt.Println("Nilai Matematika:", nilai["Matematika"]) // Ubah nilai nilai["Fisika"] = 95 fmt.Println("Nilai Fisika diubah:", nilai["Fisika"]) // Tambah data baru nilai["Biologi"] = 87 fmt.Println("Semua nilai:", nilai) // Cek key ada atau tidak biology, ada := nilai["Biologi"] if ada { fmt.Println("Nilai Biologi:", biology) } else { fmt.Println("Biologi tidak ada") } }

7.4 Menghapus Data dari Map

package main import "fmt" func main() { buah := map[string]int{ "Apel": 5000, "Mangga": 8000, "Jeruk": 3000, } fmt.Println("Sebelum hapus:", buah) // Hapus data delete(buah, "Jeruk") fmt.Println("Setelah hapus:", buah) }

7.5 Looping Map

package main import "fmt" func main() { kontak := map[string]string{ "Budi": "081234567890", "Ani": "082345678901", "Citra": "083456789012", } fmt.Println("===== DAFTAR KONTAK =====") for nama, nohp := range kontak { fmt.Printf("%s: %s\n", nama, nohp) } }

7.6 Map dengan Value Slice

package main import "fmt" func main() { // Map dengan value berupa slice nilaiMahasiswa := map[string][]int{ "Budi": {85, 90, 88}, "Ani": {90, 92, 89}, "Citra": {78, 85, 80}, } fmt.Println("===== NILAI MAHASISWA =====") for nama, nilaiList := range nilaiMahasiswa { total := 0 for _, nilai := range nilaiList { total += nilai } rataRata := float64(total) / float64(len(nilaiList)) fmt.Printf("%s: %v (Rata-rata: %.2f)\n", nama, nilaiList, rataRata) } }

7.7 Nested Map (Map dalam Map)

package main import "fmt" func main() { // Map dalam map mahasiswa := map[string]map[string]interface{}{ "Budi": { "Umur": 20, "Jurusan": "Informatika", "IPK": 3.8, }, "Ani": { "Umur": 21, "Jurusan": "Sistem Informasi", "IPK": 3.9, }, } fmt.Println("===== DATA MAHASISWA =====") for nama, data := range mahasiswa { fmt.Printf("\n%s:\n", nama) for key, value := range data { fmt.Printf(" %s: %v\n", key, value) } } }

7.8 Contoh: Sistem Inventori

package main import "fmt" func main() { // Inventori barang stok := map[string]int{ "Laptop": 10, "Mouse": 50, "Keyboard": 30, } harga := map[string]int{ "Laptop": 5000000, "Mouse": 100000, "Keyboard": 200000, } for { fmt.Println("\n===== SISTEM INVENTORI =====") fmt.Println("1. Lihat Stok") fmt.Println("2. Tambah Stok") fmt.Println("3. Kurangi Stok") fmt.Println("4. Keluar") fmt.Print("Pilih menu: ") var pilihan int fmt.Scan(&pilihan) switch pilihan { case 1: fmt.Println("\n===== DAFTAR STOK =====") for barang, jumlah := range stok { hrg := harga[barang] fmt.Printf("%s: %d unit (Rp %d)\n", barang, jumlah, hrg) } case 2: var barang string var jumlah int fmt.Print("Nama barang: ") fmt.Scan(&barang) fmt.Print("Jumlah tambah: ") fmt.Scan(&jumlah) stok[barang] += jumlah fmt.Printf("✓ %s bertambah %d unit\n", barang, jumlah) case 3: var barang string var jumlah int fmt.Print("Nama barang: ") fmt.Scan(&barang) fmt.Print("Jumlah kurang: ") fmt.Scan(&jumlah) if stok[barang] >= jumlah { stok[barang] -= jumlah fmt.Printf("✓ %s berkurang %d unit\n", barang, jumlah) } else { fmt.Println("✗ Stok tidak cukup!") } case 4: fmt.Println("Terima kasih!") return default: fmt.Println("Pilihan tidak valid!") } } }

Latihan 7

  • Soal 1: Buatlah program kamus Inggris-Indonesia menggunakan map. User bisa:
    • Tambah kata baru
    • Cari terjemahan
    • Hapus kata
    • Lihat semua kata
  • Soal 2: Buatlah program untuk menghitung frekuensi kemunculan kata dalam sebuah kalimat menggunakan map

💡 Tips Map:

  • Gunakan make(map[KeyType]ValueType) untuk membuat map kosong
  • Selalu cek apakah key ada dengan value, ok := myMap[key]
  • Gunakan delete(map, key) untuk menghapus
  • Map tidak berurutan, jangan andalkan urutan saat looping
  • Key harus comparable (bisa dibandingkan)
