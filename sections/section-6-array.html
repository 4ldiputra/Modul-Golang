6. Array dan Slice

6.1 Apa Itu Array?

Array adalah kumpulan data dengan tipe yang sama dan ukuran tetap (fixed size).

Karakteristik Array:

  • Ukuran tetap (tidak bisa berubah)
  • Tipe data harus sama
  • Index dimulai dari 0

6.2 Deklarasi Array

package main import "fmt" func main() { // Cara 1: Deklarasi dengan ukuran var angka [5]int angka[0] = 10 angka[1] = 20 angka[2] = 30 angka[3] = 40 angka[4] = 50 fmt.Println(angka) // Cara 2: Deklarasi dengan nilai langsung buah := [3]string{"Apel", "Mangga", "Jeruk"} fmt.Println(buah) // Cara 3: Ukuran otomatis warna := [...]string{"Merah", "Hijau", "Biru", "Kuning"} fmt.Println(warna) fmt.Println("Jumlah warna:", len(warna)) }

6.3 Mengakses dan Mengubah Array

package main import "fmt" func main() { nilai := [5]int{85, 90, 75, 88, 92} // Akses elemen fmt.Println("Nilai ke-1:", nilai[0]) fmt.Println("Nilai ke-3:", nilai[2]) // Ubah elemen nilai[2] = 95 fmt.Println("Setelah diubah:", nilai) // Panjang array fmt.Println("Jumlah elemen:", len(nilai)) }

6.4 Looping Array

package main import "fmt" func main() { mahasiswa := [4]string{"Budi", "Ani", "Citra", "Dedi"} // Cara 1: For biasa fmt.Println("=== FOR BIASA ===") for i := 0; i < len(mahasiswa); i++ { fmt.Printf("%d. %s\n", i+1, mahasiswa[i]) } // Cara 2: For range fmt.Println("\n=== FOR RANGE ===") for index, nama := range mahasiswa { fmt.Printf("%d. %s\n", index+1, nama) } // Cara 3: Range tanpa index fmt.Println("\n=== TANPA INDEX ===") for _, nama := range mahasiswa { fmt.Println("-", nama) } }

6.5 Array Multidimensi

package main import "fmt" func main() { // Array 2 dimensi (seperti tabel) var matrix [3][3]int = [3][3]int{ {1, 2, 3}, {4, 5, 6}, {7, 8, 9}, } fmt.Println("=== MATRIX 3x3 ===") for i := 0; i < 3; i++ { for j := 0; j < 3; j++ { fmt.Print(matrix[i][j], " ") } fmt.Println() } }

6.6 Apa Itu Slice?

Slice adalah versi dinamis dari array (ukuran bisa berubah).

Kelebihan Slice:

  • Ukuran fleksibel (bisa bertambah/berkurang)
  • Lebih sering digunakan daripada array
  • Punya fungsi bawaan seperti append

6.7 Deklarasi Slice

package main import "fmt" func main() { // Cara 1: Seperti array tapi tanpa ukuran buah := []string{"Apel", "Mangga", "Jeruk"} fmt.Println(buah) // Cara 2: Dengan make angka := make([]int, 3) // slice dengan 3 elemen angka[0] = 10 angka[1] = 20 angka[2] = 30 fmt.Println(angka) // Cara 3: Dari array arr := [5]int{1, 2, 3, 4, 5} slice := arr[1:4] // ambil dari index 1-3 fmt.Println(slice) }

6.8 Menambah Elemen (append)

package main import "fmt" func main() { hobi := []string{"Membaca", "Olahraga"} fmt.Println("Awal:", hobi) // Tambah 1 elemen hobi = append(hobi, "Musik") fmt.Println("Tambah 1:", hobi) // Tambah banyak elemen hobi = append(hobi, "Gaming", "Coding") fmt.Println("Tambah banyak:", hobi) fmt.Println("Jumlah hobi:", len(hobi)) }

6.9 Copy Slice

package main import "fmt" func main() { angka1 := []int{1, 2, 3} angka2 := make([]int, len(angka1)) copy(angka2, angka1) fmt.Println("Angka 1:", angka1) fmt.Println("Angka 2:", angka2) // Ubah angka2 tidak pengaruh angka1 angka2[0] = 999 fmt.Println("\nSetelah ubah angka2:") fmt.Println("Angka 1:", angka1) fmt.Println("Angka 2:", angka2) }

6.10 Contoh: Daftar Nilai Mahasiswa

package main import "fmt" func main() { mahasiswa := []string{} nilai := []int{} var n int fmt.Print("Berapa mahasiswa? ") fmt.Scan(&n) // Input data for i := 0; i < n; i++ { var nama string var score int fmt.Printf("\nMahasiswa %d:\n", i+1) fmt.Print("Nama : ") fmt.Scan(&nama) fmt.Print("Nilai : ") fmt.Scan(&score) mahasiswa = append(mahasiswa, nama) nilai = append(nilai, score) } // Tampilkan data fmt.Println("\n===== DAFTAR NILAI =====") total := 0 for i := 0; i < len(mahasiswa); i++ { fmt.Printf("%d. %s - %d\n", i+1, mahasiswa[i], nilai[i]) total += nilai[i] } rataRata := float64(total) / float64(len(nilai)) fmt.Printf("\nRata-rata: %.2f\n", rataRata) }
